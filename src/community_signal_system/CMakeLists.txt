cmake_minimum_required(VERSION 3.16)
project(community_signal_system)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_library(community_signal_system SHARED src/SignalSystem.cc)
target_compile_features(community_signal_system PUBLIC cxx_std_17)
target_link_libraries(community_signal_system PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
install(TARGETS community_signal_system LIBRARY DESTINATION lib)
ament_package()
